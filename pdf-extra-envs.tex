% pdf-extra-envs.tex — סביבות/חבילות נוספות ל-PDF שאינן מוגדרות אחרת בפרוייקט.
% חבילות שהמקור משתמש בהן בתוכן (למשל \alph* ברשימות) ואינן נטענות ע"י תבנית ברירת המחדל:
\usepackage{enumitem}
\usepackage{mathtools}
\makeatletter
\AtBeginDocument{%
  % הביבליוגרפיה ריקה (אין ציטוטים) → CSLReferences ריק קורס ב-LaTeX ("missing \item").
  % הופכים אותו לסביבה שקופה שסובלת ריקנות. (אם יתווספו ציטוטים בעתיד — יש להסיר תיקון זה.)
  \@ifundefined{CSLReferences}{}{\renewenvironment{CSLReferences}[2]{\par}{\par}}%
  % \begin{teacher}...\end{teacher} — חומר למורה. מוגדר כמסגרת אפורה (mdframed תומך RTL).
  \@ifundefined{teacher}{%
    \@ifundefined{newmdenv}{%
      \newenvironment{teacher}{\par\noindent\textbf{למורה:}\par}{\par}%
    }{%
      \newmdenv[linecolor=gray!55, backgroundcolor=gray!5, linewidth=0.5pt,
        leftmargin=0pt, rightmargin=0pt, innerleftmargin=8pt, innerrightmargin=8pt,
        innertopmargin=6pt, innerbottommargin=6pt, skipabove=10pt, skipbelow=10pt]{teacher}%
    }%
  }{}%
}
\makeatother
